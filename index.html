// ★V1.38: 内圧を丸めずに保存する(Math.round を廃止)。 // TPU-MINIアプリが単位(unit: "kPa" / "bar" / "psi")を送ってきた時は、その単位の値として保存し、 // メモ欄に「#車番 / セット番号 / 単位」と書く。見た目の桁はアプリと同じ(kPa 0.0 / bar 0.00 / psi 0.0)で、 // セルを選ぶと丸めていない値が見える。unit が届かない旧Core2版は従来どおり kPa 扱いで、メモに単位は足さない。 function doPost(e) { try { // 1. アクティブなシートを取得 var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); // 2. M5Core2 から送られてきた JSON 配列をパース var jsonString = e.postData.contents; var dataArray = JSON.parse(jsonString); if (!Array.isArray(dataArray) || dataArray.length === 0) { return ContentService.createTextOutput(JSON.stringify({ status: "ERROR", message: "No data found" })) .setMimeType(ContentService.MimeType.JSON); } // 配列から各タイヤのデータを整理（一括送信構造に対応） var flData = null, frData = null, rlData = null, rrData = null; var carNo = ""; // ★V1.36追加: 最初の測定〜4輪完了までの経過秒数(elapsedSec)。どのタイヤのデータにも同じ値が入っている var elapsedSec = 0; // ★V1.37追加: タイヤのセット番号(TPU-MINIアプリ第7.5版以降)。旧Core2版からは届かないので空欄のまま var tireSet = ""; // ★V1.38追加: 単位(TPU-MINIアプリが送る "kPa" / "bar" / "psi")。届かなければ空欄 = 従来どおり kPa var unit = ""; dataArray.forEach(function(item) { if (item.sessionId) { carNo = item.sessionId; // 👈 M5Core2から届いた「S-07」「S-12」などの文字をそのまま使用します } if (item.tireSet !== undefined && item.tireSet !== null && String(item.tireSet).trim() !== "") { tireSet = String(item.tireSet).trim(); } if (typeof item.elapsedSec !== "undefined" && item.elapsedSec !== null) { elapsedSec = Number(item.elapsedSec) || 0; } if (item.unit !== undefined && item.unit !== null && UNIT_FORMATS[String(item.unit).trim()]) { unit = String(item.unit).trim(); } switch(item.tireIndex) { case 0: flData = item; break; // FL case 1: rlData = item; break; // RL case 2: rrData = item; break; // RR case 3: frData = item; break; // FR } }); // ★V1.38: 内圧セルの見た目の桁(値そのものは丸めない) var numFormat = UNIT_FORMATS[unit || "kPa"]; // 代表してモード名を取得 var modeMemo = ""; dataArray.forEach(function(item) { if (item.mode === "ADJUST") { modeMemo = ""; } }); // ★V1.36追加: 経過秒数を "分:秒" 表記の文字列に変換（0秒/未送信時は空欄のまま） var elapsedStr = ""; if (elapsedSec > 0) { var mm = Math.floor(elapsedSec / 60); var ss = elapsedSec % 60; elapsedStr = mm + ":" + (ss < 10 ? "0" + ss : ss); } // 3. 次に書き込むべきブロックの「左上セル（行, 列）」を自動計算 // B2(2行目, 2列目)から開始。 // 1ブロック4列ピッチ (データ2列 + メモ1列 + 空き1列 = 4列) ➔ (b * 4) // b=0: 2列目 (B, Cが内圧、Dがメモ ➔ E列が空き) // b=1: 6列目 (F, Gが内圧、Hがメモ ➔ I列が空き) // b=2: 10列目 (J, Kが内圧、Lがメモ ➔ M列が空き) // b=3: 14列目 (N, Oが内圧、Pがメモ ➔ Q列が空き) // 縦は3行使用 + 1行空ける ＝ 4行ピッチ var startRow = 2; var startCol = 2; var targetRow = startRow; var targetCol = startCol; var found = false; // 空いているブロック位置を探索 for (var r = startRow; r < 2000; r += 4) { // 縦方向に4行ずつ進む for (var b = 0; b < 4; b++) { // 横方向に4ブロック確認 var c = startCol + (b * 4); // そのブロックの左上セル（FLの位置）が空いているか確認 if (sheet.getRange(r, c).getValue() === "") { targetRow = r; targetCol = c; found = true; break; } } if (found) break; } // タイムスタンプの生成 (西暦なし、月/日 時:分:秒 形式) var timestamp = Utilities.formatDate(new Date(), "Asia/Tokyo", "MM/dd HH:mm:ss"); // 4. 計算されたターゲット位置へ「田の字＋メモブロック」を書き込み // 1行目: FL | FR | メモ欄(車番 / セット番号 / 単位) if (flData) setPressure_(sheet, targetRow, targetCol, flData, numFormat); if (frData) setPressure_(sheet, targetRow, targetCol + 1, frData, numFormat); // ★V1.37: セット番号があれば「#007 / A3」、無ければ従来どおり「#007」 // ★V1.38: 単位が届いた時は最後に「 / psi」などを足す var memoText = modeMemo + carNo + (tireSet ? " / " + tireSet : "") + (unit ? " / " + unit : ""); // 「1-2」のような文字が日付などに自動変換されないよう、文字列として書き込む sheet.getRange(targetRow, targetCol + 2).setNumberFormat("@").setValue(memoText); // 2行目: RL | RR if (rlData) setPressure_(sheet, targetRow + 1, targetCol, rlData, numFormat); if (rrData) setPressure_(sheet, targetRow + 1, targetCol + 1, rrData, numFormat); // 3行目: タイムスタンプ（FLとFRの下の2列分を結合して値をセット） var timeRange = sheet.getRange(targetRow + 2, targetCol, 1, 2); timeRange.merge(); // セル結合を実行 timeRange.setValue(timestamp); // ★V1.36追加: 3行目・メモ列の下(空いていたセル)に「所要時間(分:秒)」を書き込み if (elapsedStr) { sheet.getRange(targetRow + 2, targetCol + 2).setValue("所要 " + elapsedStr); } // セルの見た目の調整 sheet.getRange(targetRow, targetCol, 3, 2).setHorizontalAlignment("center"); // 内圧と時間は中央揃え sheet.getRange(targetRow, targetCol + 2).setHorizontalAlignment("left"); // メモ欄は左寄せ timeRange.setHorizontalAlignment("center"); // 結合時間枠も左寄せ sheet.getRange(targetRow + 2, targetCol + 2).setHorizontalAlignment("left"); // ★V1.36追加: 所要時間欄も左寄せ // 書き込んだブロック全体（3行×3列）に黒の実線で罫線を適用 var blockRange = sheet.getRange(targetRow, targetCol, 3, 3); blockRange.setBorder(true, true, true, true, true, true, "#000000", SpreadsheetApp.BorderStyle.SOLID); // 全4ブロック分の列幅を固定値（ピクセル）で直接ガチッと固定 for (var b = 0; b < 4; b++) { var c = startCol + (b * 4); sheet.setColumnWidth(c, 60); // 左内圧列 sheet.setColumnWidth(c + 1, 60); // 右内圧列 sheet.setColumnWidth(c + 2, 160); // メモ欄列(1行目=車番／3行目=所要時間) sheet.setColumnWidth(c + 3, 25); // ブロック間の空き列 } return ContentService.createTextOutput(JSON.stringify({ status: "SUCCESS", row: targetRow, col: targetCol, elapsedSec: elapsedSec, unit: unit || "kPa" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "ERROR", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // ★V1.38追加: 単位ごとの見た目の桁(アプリと同じ)。値そのものは丸めずに保存する var UNIT_FORMATS = { "kPa": "0.0", "bar": "0.00", "psi": "0.0" }; // ★V1.38追加: 内圧を丸めずに書き込み、見た目の桁だけ単位に合わせる function setPressure_(sheet, row, col, data, numFormat) { var v = Number(data.pressure); var cell = sheet.getRange(row, col); cell.setNumberFormat(numFormat); cell.setValue(isNaN(v) ? data.pressure : v); }
